// Wet-only impulse measurements, with fade-in completed before excitation.
float[] Impulse(ReverbCharacter type, int decay, float amplitude = .5f)
{
    var a = new ReverbEffect(48000); a.Configure(true, type, 85, decay, 55, 18, 45, 60);
    for (int i = 0; i < 4800; i++) a.Process(0, 0, out _, out _);
    var result = new float[48000 * 4];
    for (int i = 0; i < result.Length; i++)
    {
        float x = i == 0 ? amplitude : 0;
        a.Process(x, x, out float l, out _); result[i] = l - x;
    }
    return result;
}
double Energy(float[] x, double start, double end) => x.Skip((int)(start * 48000))
    .Take((int)((end - start) * 48000)).Sum(v => (double)v * v);
double Tail(float[] x, double threshold)
{
    var e = new double[x.Length]; double sum = 0;
    for (int i = x.Length - 1; i >= 0; i--) { sum += (double)x[i] * x[i]; e[i] = sum; }
    return Array.FindIndex(e, v => v < sum * threshold) / 48000.0;
}
double Irregularity(float[] x)
{
    // Absence of periodic repetitions in the first 250 ms, measured by autocorrelation.
    double largest = 0;
    for (int lag = 384; lag <= 2400; lag += 8)
    {
        double cross = 0, e1 = 0, e2 = 0;
        for (int i = 1920; i < 12000; i++)
        { double a = x[i], b = x[i + lag]; cross += a * b; e1 += a * a; e2 += b * b; }
        largest = Math.Max(largest, Math.Abs(cross) / Math.Sqrt(Math.Max(1e-30, e1 * e2)));
    }
    return 1 - largest;
}

// Dynamics: compare normalized early response to strong and soft plucked chords.
double PluckResponse(float amplitude, bool soft)
{
    var a = new ReverbEffect(48000); a.Configure(true, ReverbCharacter.Spring, 85, 48, 55, 18, 45, 60);
    double energy = 0;
    for (int i = -4800; i < 24000; i++)
    {
        float x = i < 0 ? 0 : Signal(i, 48000) * amplitude * MathF.Exp(-i / 5500f)
            * (soft ? Math.Min(1, i / 2400f) : Math.Min(1, i / 24f));
        a.Process(x, x, out float l, out _);
        if (i >= 864 && i < 4800) energy += (l - x) * (l - x);
    }
    return energy / (amplitude * amplitude);
}
